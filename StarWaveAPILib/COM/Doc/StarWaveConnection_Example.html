 StarWaveAPI.StarWaveConnection FConnection = null;
//--------------------------------------------------------------------------------------------------------------------------------------------------
public MainForm()
{
    InitializeComponent();
    // 產生StarWaveConnection物件
    FConnection = new StarWaveAPI.StarWaveConnection();
    // Handle 連線,斷線,商品下載完成的事件
    FConnection.OnConnected += new StarWaveAPI.IStarWaveConnectionEvents_OnConnectedEventHandler(FConnection_OnConnected);
    FConnection.OnDisconnected += new StarWaveAPI.IStarWaveConnectionEvents_OnDisconnectedEventHandler(FConnection_OnDisconnected);
    FConnection.OnContractDownloadCompleted += new StarWaveAPI.IStarWaveConnectionEvents_OnContractDownloadCompletedEventHandler(FConnection_OnContractDownloadCompleted);
    // Handel 行情異動產生的事件.
    FConnection.OnRecover += new StarWaveAPI.IStarWaveConnectionEvents_OnRecoverEventHandler(FConnection_OnRecover);
    FConnection.OnOrderBookData += new StarWaveAPI.IStarWaveConnectionEvents_OnOrderBookDataEventHandler(FConnection_OnOrderBookData);
    FConnection.OnMatchInfo += new StarWaveAPI.IStarWaveConnectionEvents_OnMatchInfoEventHandler(FConnection_OnMatchInfo);
    FConnection.OnDayHighLow += new StarWaveAPI.IStarWaveConnectionEvents_OnDayHighLowEventHandler(FConnection_OnDayHighLow);
    FConnection.OnUnderlyingIndex += new StarWaveAPI.IStarWaveConnectionEvents_OnUnderlyingIndexEventHandler(FConnection_OnUnderlyingIndex);
    FConnection.OnTotalMatch += new StarWaveAPI.IStarWaveConnectionEvents_OnTotalMatchEventHandler(FConnection_OnTotalMatch);
    FConnection.OnSumOfOrder += new StarWaveAPI.IStarWaveConnectionEvents_OnSumOfOrderEventHandler(FConnection_OnSumOfOrder);
    FConnection.OnOpeningInfo += new StarWaveAPI.IStarWaveConnectionEvents_OnOpeningInfoEventHandler(FConnection_OnOpeningInfo);
    FConnection.OnClosingMarketData += new StarWaveAPI.IStarWaveConnectionEvents_OnClosingMarketDataEventHandler(FConnection_OnClosingMarketData);
}
//--------------------------------------------------------------------------------------------------------------------------------------------------
void FOrderConnection_OnConnected()
{ 
    // 與StarWave Server連線成功.
    System.Console.WriteLine("FConnection_OnConnected");
    // 向Server要所有可註冊行情的商品.
    FConnection.DownloadContract();
}
//--------------------------------------------------------------------------------------------------------------------------------------------------
void FOrderConnection_OnDisconnected()
{ 
    // 與StarWave Server斷線.
     System.Console.WriteLine("FConnection_OnDisconnected");
}

void FConnection_OnContractDownloadCompleted(int ExchangeCount)
{
    // 商品下載完畢.
    for (int i = 0; i < ExchangeCount; i++) 
    {
        // 取得每一個交易所的交易所代碼.
        string Exchange = FConnection.GetExchange(i);
        TreeNode ExNode = null;
        if( CommodTreeView.Nodes.ContainsKey(Exchange))
        {
            TreeNode[] result = CommodTreeView.Nodes.Find(Exchange, false);
            ExNode = result[0];
        }
        else
            ExNode = CommodTreeView.Nodes.Add(Exchange, Exchange);
        // 取得某一交易所下,所有的商品基本資料
        for (int j = 0; j < FConnection.SymbolCount(Exchange); j++)
        {  
            string Symbol = FConnection.GetSymbol(Exchange, j);
            StarWaveAPI.SecurityDefinition Def = new StarWaveAPI.SecurityDefinition();
            FConnection.GetSecurityDefinition(Exchange, Symbol, Def);
            if (Def != null)
            {
                TreeNode MarketNode, ProdNode, MaturityDateNode ,SymbolNode;
                if ( ExNode.Nodes.ContainsKey(Def.Market.ToString()))
                {
                    TreeNode[] node = ExNode.Nodes.Find(Def.Market.ToString(), false);
                    MarketNode = node[0];
                }
                else
                    MarketNode = ExNode.Nodes.Add(Def.Market.ToString(), Def.Market.ToString());
                if( MarketNode.Nodes.ContainsKey(Def.ProductID))
                {
                    TreeNode[] node = MarketNode.Nodes.Find(Def.ProductID, false);
                    ProdNode = node[0];
                }
                else
                    ProdNode = MarketNode.Nodes.Add(Def.ProductID, Def.ProductID);
                if (Def.MaturityDate != null)
                {
                    if( ProdNode.Nodes.ContainsKey(Def.MaturityDate))
                    {
                        TreeNode[] node = ProdNode.Nodes.Find(Def.MaturityDate, false);
                        MaturityDateNode = node[0];
                    }
                    else
                        MaturityDateNode = ProdNode.Nodes.Add(Def.MaturityDate, Def.MaturityDate);
                }
                else
                     MaturityDateNode = null;
                if (MaturityDateNode != null)
                    SymbolNode = MaturityDateNode.Nodes.Add(Def.Symbol, Def.Symbol);
                else
                    SymbolNode = ProdNode.Nodes.Add(Def.Symbol, Def.Symbol);
            }
        }
    }
    CommodTreeView.Sort();
}<}

void FConnection_OnRecover(StarWaveAPI.Recover Recover)
{
    string TableKey = string.Format("{0}.{1}", Recover.Exchange, Recover.Symbol);
    DepthForm UpdaateForm = (DepthForm)FTable[TableKey];
    Console.WriteLine( "OnRecover:[{0}]", TableKey );
    if (UpdaateForm != null)
        UpdaateForm.OnRecover(Recover);
}
//--------------------------------------------------------------------------------------------------------------------------------------------------
void FConnection_OnOrderBookData(StarWaveAPI.OrderBook OrderBookData)
{ {
    string TableKey = string.Format("{0}.{1}", OrderBookData.Exchange, OrderBookData.Symbol);
    DepthForm UpdaateForm = (DepthForm)FTable[TableKey];
    if (UpdaateForm != null)
        UpdaateForm.Update(OrderBookData);
